# Match-flow probe scenarios, run by tools/net/flow_probes.py.
#   scenario <name>
#   peer <name> <t>:<verb>[=<arg>];...            the plan tools/net/flow_probe.gd follows
#   expect <peer> <t0> <t1> all|any|none <regex>  over that peer's PROBE lines in the window
#   exit <peer> <code>                            when a peer is meant to die (default 0)
#   max_errors <peer> <count>                     engine error lines the peer may log
